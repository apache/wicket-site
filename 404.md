// Suggests the new home of an old address, from the part of the path that moved.
(function () {
    var path = window.location.pathname.replace('{{ site.baseurl }}', '');
    // A page name under /docs/ is searched for, so a renamed page still turns up.
    function search(m) {
        var words = m[1].replace(/\.html$/, '').replace(/[-_]+/g, ' ');
        return ['/docs/a-z.html?q=' + encodeURIComponent(words), 'in the documentation, under \u201c' + words + '\u201d'];
    }
    var moves = [
        [/^\/(?:start|versions)\//, function () { return ['/versions/', 'on the release lines page']; }],
        [/^\/docs\/(?:topics\/)?([^/]+)$/, search],
        [/^\/docs\//, function () { return ['/docs/', 'in the documentation']; }],
        [/^\/learn\//, function () { return ['/learn/', 'on Learn Wicket']; }],
        [/^\/help\//, function () { return ['/community/', 'on the community page']; }],
        [/^\/(news\/)?20[0-9][0-9]\//, function () { return ['/news/', 'on the blog']; }]
    ];
    for (var i = 0; i < moves.length; i++) {
        var m = path.match(moves[i][0]);
        if (!m) continue;
        var target = moves[i][1](m);
        var p = document.querySelector('.notfound-suggest');
        var a = p.querySelector('a');
        a.href = '{{ site.baseurl }}' + target[0];
        a.textContent = target[1];
        p.hidden = false;
        break;
    }
})();
</script>
